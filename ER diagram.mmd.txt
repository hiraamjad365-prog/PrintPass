    USER {
        int user_id PK
        varchar full_name
        varchar email
        varchar password_hash
        varchar phone
        varchar role
        varchar account_status
        datetime created_at
    }

    STUDENT {
        int student_id PK
        int user_id FK
        varchar registration_no
        varchar department
        int semester
    }

    STAFF {
        int staff_id PK
        int user_id FK
        varchar staff_role
        date joining_date
    }

    CATEGORY {
        int category_id PK
        varchar category_name
        varchar description
    }

    PRODUCT {
        int product_id PK
        int category_id FK
        varchar product_name
        decimal selling_price
        int stock_quantity
        int reorder_level
        varchar product_status
    }

    BOOK_DETAILS {
        int product_id PK, FK
        varchar ISBN
        varchar author
        varchar publisher
        varchar edition
    }

    ORDERS {
        int order_id PK
        int student_id FK
        int processed_by FK
        varchar order_number
        varchar order_type
        datetime order_date
        varchar order_status
        decimal total_amount
    }

    ORDER_ITEM {
        int order_item_id PK
        int order_id FK
        int product_id FK
        int quantity
        decimal unit_price
        decimal subtotal
    }

    INVENTORY_TRANSACTION {
        int transaction_id PK
        int product_id FK
        int staff_id FK
        int order_item_id FK
        varchar transaction_type
        int quantity_change
        datetime transaction_date
    }

    PRICE_RULE {
        int price_rule_id PK
        varchar service_type
        varchar color_mode
        varchar paper_size
        decimal rate_per_page
        date effective_from
        boolean is_active
    }

    PRINT_JOB {
        int print_job_id PK
        int order_id FK
        int price_rule_id FK
        varchar file_name
        varchar file_path
        int total_pages
        int number_of_copies
        varchar print_type
        decimal rate_per_page
        decimal total_cost
        varchar print_status
    }

    QUEUE_TICKET {
        int queue_id PK
        int print_job_id FK
        varchar token_number
        date queue_date
        varchar queue_status
        datetime issued_at
        datetime completed_at
    }

    PAYMENT {
        int payment_id PK
        int order_id FK
        int verified_by FK
        decimal amount
        varchar payment_method
        varchar transaction_reference
        varchar payment_status
        datetime payment_date
    }

    RECEIPT {
        int receipt_id PK
        int payment_id FK
        varchar receipt_number
        datetime issue_date
        varchar receipt_path
    }

    NOTIFICATION {
        int notification_id PK
        int user_id FK
        int print_job_id FK
        varchar notification_title
        varchar notification_message
        varchar notification_type
        boolean is_read
        datetime created_at
    }

    ACTIVITY_LOG {
        int log_id PK
        int user_id FK
        varchar action
        varchar entity_name
        int record_id
        datetime activity_time
    }
